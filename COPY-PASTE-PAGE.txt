import Link from "next/link";
import {
  ArrowLeft,
  CheckCircle2,
  Download,
  FileCheck2,
  FileImage,
  FilePdf,
  Info,
  Loader2,
  LockKeyhole,
  ShieldCheck,
  Sparkles,
  UploadCloud,

  Gauge,

  AlertTriangle,
} from "lucide-react";

type PreviewPage = {
  pageNumber: number;
  dataUrl: string;
};

type QualityMode = "auto" | "high" | "balanced" | "strong";
type EngineMode = "low" | "balanced" | "high";

type CompressionProfile = {
  label: string;
  mode: EngineMode;
  imageQuality: number;
  maxImageDimension: number;
  targetDpi: number;
};

type CompressionResult = {
  blob: Blob;
  outputSize: number;
  savedBytes: number;
  savedPercent: number;
  reachedTarget: boolean;
  profileLabel: string;
  imageQuality: number;
  message: string;
};

const TARGET_MIN = 5;
const TARGET_MAX = 90;
const MB = 1024 * 1024;

const PROFILES: Record<Exclude<QualityMode, "auto">, CompressionProfile> = {
  high: {
    label: "High quality",
    mode: "low",
    imageQuality: 0.85,
    maxImageDimension: 2400,
    targetDpi: 220,
  },
  balanced: {
    label: "Balanced",
    mode: "balanced",
    imageQuality: 0.72,
    maxImageDimension: 1800,
    targetDpi: 160,
  },
  strong: {
    label: "Smaller file",
    mode: "high",
    imageQuality: 0.55,
    maxImageDimension: 1400,
    targetDpi: 130,
  },
};

function formatBytes(bytes: number): string {
  if (!Number.isFinite(bytes) || bytes <= 0) return "0 Bytes";
  const units = ["Bytes", "KB", "MB", "GB"];
  const index = Math.min(Math.floor(Math.log(bytes) / Math.log(1024)), units.length - 1);
  return `${(bytes / Math.pow(1024, index)).toFixed(index === 0 ? 0 : 2)} ${units[index]}`;
}

function getOutputName(name: string): string {
  const dot = name.lastIndexOf(".");
  const base = dot > 0 ? name.slice(0, dot) : name;
  return `${base}-compressed.pdf`;
}

async function renderPdfThumbnails(file: File): Promise<PreviewPage[]> {
  const pdfjs = await import("pdfjs-dist/legacy/build/pdf.mjs");
  pdfjs.GlobalWorkerOptions.workerSrc =
    `https://cdn.jsdelivr.net/npm/pdfjs-dist@${pdfjs.version}/build/pdf.worker.min.mjs`;

  const buffer = await file.arrayBuffer();
  const loadingTask = pdfjs.getDocument({
    data: new Uint8Array(buffer.slice(0)),
  });

  try {
    const pdf = await loadingTask.promise;
    const previews: PreviewPage[] = [];

    for (let pageNumber = 1; pageNumber <= pdf.numPages; pageNumber += 1) {
      const page = await pdf.getPage(pageNumber);
      const baseViewport = page.getViewport({ scale: 1 });
      const scale = Math.min(190 / baseViewport.width, 0.42);
      const viewport = page.getViewport({ scale: Math.max(scale, 0.18) });
      const canvas = document.createElement("canvas");
      const context = canvas.getContext("2d", { alpha: false });

      if (!context) {
        throw new Error("Your browser could not create a preview canvas.");
      }

      canvas.width = Math.ceil(viewport.width);
      canvas.height = Math.ceil(viewport.height);

      try {
        await page.render({ canvas, canvasContext: context, viewport }).promise;
        previews.push({
          pageNumber,
          dataUrl: canvas.toDataURL("image/jpeg", 0.78),
        });
      } finally {
        // Release thumbnail canvas memory even if rendering fails.
        canvas.width = 1;
        canvas.height = 1;
        page.cleanup();
      }
    }

    return previews;
  } finally {
    // PDF.js exposes destroy() reliably on the loading task.
    // Do not call pdf.destroy(); some installed PDF.js builds do not expose it.
    try {
      await loadingTask.destroy();
    } catch (cleanupError) {
      console.warn("PDF.js cleanup warning:", cleanupError);
    }
  }
}

export default function CompressPdfPage() {
  const inputRef = React.useRef<HTMLInputElement>(null);
  const [file, setFile] = React.useState<File | null>(null);
  const [pageCount, setPageCount] = React.useState(0);
  const [pages, setPages] = React.useState<PreviewPage[]>([]);
  const [targetPercent, setTargetPercent] = React.useState(40);
  const [qualityMode, setQualityMode] = React.useState<QualityMode>("auto");
  const [isDragging, setIsDragging] = React.useState(false);
  const [isLoading, setIsLoading] = React.useState(false);
  const [isCompressing, setIsCompressing] = React.useState(false);
  const [progressLabel, setProgressLabel] = React.useState("");
  const [error, setError] = React.useState("");
  const [result, setResult] = React.useState<CompressionResult | null>(null);
  const [attemptedStronger, setAttemptedStronger] = React.useState(false);

  const targetBytes = file ? Math.round(file.size * (1 - targetPercent / 100)) : 0;

  const clearCurrentFile = React.useCallback(() => {
    setFile(null);
    setPageCount(0);
    setPages([]);
    setResult(null);
    setError("");
    setProgressLabel("");
    setAttemptedStronger(false);
    if (inputRef.current) inputRef.current.value = "";
  }, []);

  const loadFile = React.useCallback(async (selectedFile: File) => {
    setError("");
    setResult(null);
    setAttemptedStronger(false);

    if (selectedFile.type !== "application/pdf" && !selectedFile.name.toLowerCase().endsWith(".pdf")) {
      setError("Please upload a PDF file only.");
      return;
    }
    if (!selectedFile.size) {
      setError("This file is empty. Please choose another PDF.");
      return;
    }
    if (selectedFile.size > 150 * MB) {
      setError("This PDF is larger than 150 MB. Please try a smaller file.");
      return;
    }

    setIsLoading(true);
    setProgressLabel("Reading PDF and creating page previews…");
    try {
      const previewPages = await renderPdfThumbnails(selectedFile);
      setFile(selectedFile);
      setPageCount(previewPages.length);
      setPages(previewPages);
    } catch (cause) {
      const message = cause instanceof Error ? cause.message : "The PDF could not be opened.";
      setError(message.toLowerCase().includes("password")
        ? "This PDF is password-protected. Please upload an unlocked PDF."
        : `Unable to read this PDF. ${message}`);
      setFile(null);
      setPageCount(0);
      setPages([]);
    } finally {
      setIsLoading(false);
      setProgressLabel("");
    }
  }, []);

  const handleFileInput = (event: React.ChangeEvent<HTMLInputElement>) => {
    const selected = event.target.files?.[0];
    if (selected) void loadFile(selected);
  };

  const handleDrop = (event: React.DragEvent<HTMLDivElement>) => {
    event.preventDefault();
    setIsDragging(false);
    const selected = event.dataTransfer.files?.[0];
    if (selected) void loadFile(selected);
  };

  const compressPdf = async (forceStrong = false) => {
    if (!file) return;

    setError("");
    setResult(null);
    setIsCompressing(true);
    setAttemptedStronger(forceStrong);

    try {
      setProgressLabel("Loading the PDF image-compression engine…");
      const { compressPDF } = await import("@fileslim/compress");

      let profiles: CompressionProfile[];
      if (forceStrong || qualityMode === "strong") {
        profiles = [PROFILES.strong];
      } else if (qualityMode === "high") {
        profiles = [PROFILES.high];
      } else if (qualityMode === "balanced") {
        profiles = [PROFILES.balanced];
      } else {
        // Auto mode always tries the highest-quality profile first. It moves
        // to balanced only if the requested target was not reached.
        profiles = [PROFILES.high, PROFILES.balanced];
      }

      let best: { blob: Blob; profile: CompressionProfile } | null = null;
      let reachedTarget = false;

      for (let index = 0; index < profiles.length; index += 1) {
        const profile = profiles[index];
        setProgressLabel(
          profiles.length > 1
            ? `Trying ${profile.label.toLowerCase()} image settings (${index + 1}/${profiles.length})…`
            : `Compressing with ${profile.label.toLowerCase()} settings…`,
        );

        const compressed = await compressPDF(file, {
          mode: profile.mode,
          imageQuality: profile.imageQuality,
          maxImageDimension: profile.maxImageDimension,
          targetDpi: profile.targetDpi,
          stripMetadata: false,
          onProgress: (phase: string, percent: number) => {
            const safePercent = Number.isFinite(percent) ? ` ${Math.round(percent)}%` : "";
            setProgressLabel(`${profile.label}: ${phase}${safePercent}`);
          },
        });

        if (!(compressed?.blob instanceof Blob) || compressed.blob.size === 0) {
          throw new Error("The compression engine did not return a valid PDF file.");
        }

        if (compressed.blob.size < file.size && (!best || compressed.blob.size < best.blob.size)) {
          best = { blob: compressed.blob, profile };
        }

        if (compressed.blob.size <= targetBytes) {
          // This is the first/highest-quality profile that reaches the target.
          best = { blob: compressed.blob, profile };
          reachedTarget = true;
          break;
        }
      }

      if (!best) {
        throw new Error(
          "This PDF did not become smaller with the selected quality settings. Try Balanced or Stronger compression, or keep the original file.",
        );
      }

      const outputSize = best.blob.size;
      const savedBytes = Math.max(0, file.size - outputSize);
      const savedPercent = (savedBytes / file.size) * 100;
      const targetMet = outputSize <= targetBytes || reachedTarget;
      let message = targetMet
        ? `The target was reached using ${best.profile.label.toLowerCase()} settings. Text and vector elements are retained where supported; embedded images were optimized.`
        : `The best result with the selected quality limit is ${formatBytes(outputSize)}. The requested target was not reached, so the tool avoided applying an even stronger image reduction automatically.`;

      if (outputSize >= file.size) {
        message = "Compression did not reduce this file. The original has been kept unchanged.";
      }

      setResult({
        blob: best.blob,
        outputSize,
        savedBytes,
        savedPercent,
        reachedTarget: targetMet,
        profileLabel: best.profile.label,
        imageQuality: best.profile.imageQuality,
        message,
      });
    } catch (cause) {
      const message = cause instanceof Error ? cause.message : "Compression could not be completed.";
      setError(
        `Unable to compress this PDF. It may be encrypted, damaged, or contain features the compression engine cannot process. ${message}`,
      );
    } finally {
      setIsCompressing(false);
      setProgressLabel("");
    }
  };

  const downloadResult = () => {
    if (!result || !file) return;
    const url = URL.createObjectURL(result.blob);
    const anchor = document.createElement("a");
    anchor.href = url;
    anchor.download = getOutputName(file.name);
    document.body.appendChild(anchor);
    anchor.click();
    anchor.remove();
    window.setTimeout(() => URL.revokeObjectURL(url), 1500);
  };

  const rangePercent = Math.max(0, Math.min(100, (targetPercent / TARGET_MAX) * 100));

  return (
    <main className="min-h-screen bg-[#f7f8fc] pb-16 text-slate-900 dark:bg-[#0c1020] dark:text-slate-100">
      <div className="mx-auto w-full max-w-7xl px-4 pt-6 sm:px-6 lg:px-8">
        <section className="relative overflow-hidden rounded-[26px] bg-gradient-to-br from-[#32247f] via-[#5542c7] to-[#2676c8] px-6 py-7 text-white shadow-[0_18px_55px_rgba(73,66,174,0.20)] sm:px-9 sm:py-9">
          <div className="pointer-events-none absolute -right-12 -top-24 h-64 w-64 rounded-full bg-white/10 blur-2xl" />
          <div className="pointer-events-none absolute -bottom-28 left-1/3 h-64 w-64 rounded-full bg-cyan-300/15 blur-3xl" />
          <div className="relative z-10 flex items-start justify-between gap-4">
            <div className="max-w-2xl">
              <div className="mb-3 inline-flex items-center gap-2 rounded-full border border-white/20 bg-white/10 px-3 py-1 text-xs font-semibold tracking-wide text-indigo-50">
                <Sparkles className="h-3.5 w-3.5" /> CONVIA PDF TOOLS
              </div>
              <h1 className="text-3xl font-bold tracking-tight sm:text-4xl">Compress PDF</h1>
              <p className="mt-2 max-w-xl text-sm leading-6 text-indigo-100 sm:text-base">
                Reduce file size by optimizing embedded images while keeping text and vector content intact wherever the PDF allows.
              </p>
            </div>
            <Link href="/" className="relative z-20 inline-flex shrink-0 items-center gap-2 rounded-full border border-white/25 bg-white/10 px-4 py-2 text-sm font-semibold text-white transition hover:bg-white/20">
              <ArrowLeft className="h-4 w-4" /> Back
            </Link>
          </div>
          <div className="relative z-10 mt-6 flex flex-wrap gap-2 text-xs font-medium text-indigo-50">
            <span className="inline-flex items-center gap-1.5 rounded-full bg-white/10 px-3 py-1.5"><ShieldCheck className="h-3.5 w-3.5" /> Quality-controlled image compression</span>
            <span className="inline-flex items-center gap-1.5 rounded-full bg-white/10 px-3 py-1.5"><LockKeyhole className="h-3.5 w-3.5" /> Browser-side processing</span>
            <span className="inline-flex items-center gap-1.5 rounded-full bg-white/10 px-3 py-1.5"><FileImage className="h-3.5 w-3.5" /> Text/vector content retained where supported</span>
          </div>
        </section>

        <div className="mt-7 grid gap-6 lg:grid-cols-[minmax(0,1.15fr)_minmax(330px,0.85fr)]">
          <section className="min-w-0 rounded-3xl border border-slate-200 bg-white p-5 shadow-sm dark:border-slate-800 dark:bg-[#141a2c] sm:p-6">
            <div className="mb-4 flex items-center justify-between gap-3">
              <div>
                <p className="text-xs font-bold uppercase tracking-[0.18em] text-indigo-600 dark:text-indigo-300">Step 01</p>
                <h2 className="mt-1 text-xl font-bold">Upload your PDF</h2>
              </div>
              {file && (
                <button type="button" onClick={clearCurrentFile} className="inline-flex items-center gap-1 rounded-lg px-3 py-2 text-xs font-semibold text-slate-500 transition hover:bg-slate-100 hover:text-rose-600 dark:hover:bg-slate-800">
                  <X className="h-4 w-4" /> Remove
                </button>
              )}
            </div>

            {!file ? (
              <div
                role="button"
                tabIndex={0}
                onClick={() => inputRef.current?.click()}
                onKeyDown={(event) => {
                  if (event.key === "Enter" || event.key === " ") {
                    event.preventDefault();
                    inputRef.current?.click();
                  }
                }}
                onDragOver={(event) => { event.preventDefault(); setIsDragging(true); }}
                onDragLeave={() => setIsDragging(false)}
                onDrop={handleDrop}
                className={`flex min-h-[250px] cursor-pointer flex-col items-center justify-center rounded-2xl border-2 border-dashed px-5 py-9 text-center transition ${isDragging ? "border-indigo-500 bg-indigo-50 dark:bg-indigo-950/40" : "border-slate-300 bg-slate-50/80 hover:border-indigo-400 hover:bg-indigo-50/60 dark:border-slate-700 dark:bg-slate-900/40 dark:hover:bg-indigo-950/20"}`}
              >
                <div className="mb-4 grid h-16 w-16 place-items-center rounded-2xl bg-gradient-to-br from-indigo-100 to-blue-100 text-indigo-600 dark:from-indigo-950 dark:to-blue-950 dark:text-indigo-300"><UploadCloud className="h-8 w-8" /></div>
                <h3 className="text-base font-bold">Drop your PDF here</h3>
                <p className="mt-1 text-sm text-slate-500 dark:text-slate-400">or click to browse from your device</p>
                <span className="mt-4 rounded-full bg-indigo-600 px-5 py-2.5 text-sm font-semibold text-white shadow-lg shadow-indigo-600/20">Choose PDF file</span>
                <p className="mt-4 text-xs text-slate-400">PDF only · Maximum 150 MB</p>
              </div>
            ) : (
              <div className="rounded-2xl border border-indigo-100 bg-gradient-to-br from-indigo-50/80 to-blue-50/70 p-4 dark:border-indigo-900/60 dark:from-indigo-950/30 dark:to-blue-950/20 sm:p-5">
                <div className="flex items-start gap-3">
                  <div className="grid h-12 w-12 shrink-0 place-items-center rounded-xl bg-white text-rose-500 shadow-sm dark:bg-slate-800"><FilePdf className="h-7 w-7" /></div>
                  <div className="min-w-0 flex-1">
                    <p className="break-all text-sm font-bold text-slate-800 dark:text-slate-100">{file.name}</p>
                    <p className="mt-1 text-xs text-slate-500 dark:text-slate-400">Original file · {formatBytes(file.size)}</p>
                  </div>
                  <span className="inline-flex shrink-0 items-center gap-1 rounded-full bg-emerald-100 px-2.5 py-1 text-[11px] font-bold text-emerald-700 dark:bg-emerald-950/60 dark:text-emerald-300"><CheckCircle2 className="h-3 w-3" /> Loaded</span>
                </div>
                <div className="mt-5 grid grid-cols-2 gap-3">
                  <div className="rounded-xl border border-white/80 bg-white/80 p-3 dark:border-slate-700 dark:bg-slate-900/50">
                    <p className="text-[11px] font-semibold uppercase tracking-wider text-slate-500">Original size</p>
                    <p className="mt-1 text-lg font-bold tabular-nums">{formatBytes(file.size)}</p>
                  </div>
                  <div className="rounded-xl border border-white/80 bg-white/80 p-3 dark:border-slate-700 dark:bg-slate-900/50">
                    <p className="text-[11px] font-semibold uppercase tracking-wider text-slate-500">Total pages</p>
                    <p className="mt-1 text-lg font-bold tabular-nums">{pageCount}</p>
                  </div>
                </div>
              </div>
            )}

            <input ref={inputRef} type="file" accept="application/pdf,.pdf" onChange={handleFileInput} className="hidden" />
            {isLoading && <div className="mt-4 flex items-center gap-2 rounded-xl bg-indigo-50 px-4 py-3 text-sm font-medium text-indigo-700 dark:bg-indigo-950/40 dark:text-indigo-200"><Loader2 className="h-4 w-4 animate-spin" /> {progressLabel}</div>}
          </section>

          <section className="rounded-3xl border border-slate-200 bg-white p-5 shadow-sm dark:border-slate-800 dark:bg-[#141a2c] sm:p-6">
            <p className="text-xs font-bold uppercase tracking-[0.18em] text-indigo-600 dark:text-indigo-300">Step 02</p>
            <h2 className="mt-1 text-xl font-bold">Choose target size</h2>
            <p className="mt-2 text-sm leading-6 text-slate-500 dark:text-slate-400">Set how much smaller you want the PDF to become. The target is a goal; the actual result depends on the images and content inside your PDF.</p>

            <div className="mt-5 rounded-2xl bg-slate-50 p-4 dark:bg-slate-900/60">
              <div className="flex items-center justify-between gap-3">
                <label htmlFor="target-percent" className="text-sm font-bold">Target reduction</label>
                <div className="relative w-28">
                  <input
                    id="target-percent"
                    type="number"
                    min={TARGET_MIN}
                    max={TARGET_MAX}
                    value={targetPercent}
                    onChange={(event) => {
                      const value = Number(event.target.value);
                      if (Number.isFinite(value)) setTargetPercent(Math.max(TARGET_MIN, Math.min(TARGET_MAX, value)));
                    }}
                    className="w-full rounded-xl border border-slate-200 bg-white px-3 py-2 pr-8 text-right text-lg font-bold tabular-nums outline-none focus:border-indigo-500 focus:ring-2 focus:ring-indigo-100 dark:border-slate-700 dark:bg-slate-800 dark:focus:ring-indigo-950"
                  />
                  <span className="pointer-events-none absolute right-3 top-1/2 -translate-y-1/2 text-sm font-bold text-slate-500">%</span>
                </div>
              </div>
              <input
                aria-label="Target size reduction percentage"
                type="range"
                min={TARGET_MIN}
                max={TARGET_MAX}
                value={targetPercent}
                onChange={(event) => setTargetPercent(Number(event.target.value))}
                className="mt-5 h-2 w-full cursor-pointer appearance-none rounded-full bg-slate-200 accent-indigo-600 dark:bg-slate-700"
                style={{ background: `linear-gradient(to right, #6554d9 ${rangePercent}%, #e2e8f0 ${rangePercent}%)` }}
              />
              <div className="mt-2 flex justify-between text-[11px] text-slate-400"><span>5%</span><span>45%</span><span>90%</span></div>

              {file && (
                <div className="mt-4 grid grid-cols-2 gap-2">
                  <div className="rounded-xl border border-slate-200 bg-white p-3 dark:border-slate-700 dark:bg-slate-800">
                    <p className="text-[10px] font-semibold uppercase tracking-wide text-slate-500">Original size</p>
                    <p className="mt-1 text-sm font-bold">{formatBytes(file.size)}</p>
                  </div>
                  <div className="rounded-xl border border-indigo-200 bg-indigo-50 p-3 dark:border-indigo-900 dark:bg-indigo-950/50">
                    <p className="text-[10px] font-semibold uppercase tracking-wide text-indigo-700 dark:text-indigo-300">Target output</p>
                    <p className="mt-1 text-sm font-bold text-indigo-800 dark:text-indigo-200">{formatBytes(targetBytes)}</p>
                  </div>
                </div>
              )}
              <div className="mt-4 grid grid-cols-4 gap-2">
                {[15, 25, 40, 50].map((value) => (
                  <button key={value} type="button" onClick={() => setTargetPercent(value)} className={`rounded-xl border px-2 py-2 text-center text-xs font-bold transition ${targetPercent === value ? "border-indigo-500 bg-indigo-600 text-white shadow-md shadow-indigo-600/20" : "border-slate-200 bg-white text-slate-600 hover:border-indigo-300 dark:border-slate-700 dark:bg-slate-800 dark:text-slate-300"}`}>
                    {value}%
                  </button>
                ))}
              </div>
            </div>

            <div className="mt-5">
              <p className="text-sm font-bold">Quality preference</p>
              <div className="mt-2 grid gap-2">
                <button type="button" onClick={() => setQualityMode("auto")} className={`rounded-xl border p-3 text-left transition ${qualityMode === "auto" ? "border-indigo-500 bg-indigo-50 dark:bg-indigo-950/40" : "border-slate-200 hover:border-indigo-300 dark:border-slate-700"}`}>
                  <div className="flex items-center justify-between gap-2"><span className="text-sm font-bold">Auto · Preserve quality first</span>{qualityMode === "auto" && <CheckCircle2 className="h-4 w-4 text-indigo-600" />}</div>
                  <p className="mt-1 text-xs text-slate-500 dark:text-slate-400">Tries high quality first, then balanced only if needed. Recommended.</p>
                </button>
                <div className="grid grid-cols-3 gap-2">
                  {([
                    ["high", "High"],
                    ["balanced", "Balanced"],
                    ["strong", "Smaller"],
                  ] as const).map(([value, label]) => (
                    <button key={value} type="button" onClick={() => setQualityMode(value)} className={`rounded-xl border px-2 py-2.5 text-center text-xs font-bold transition ${qualityMode === value ? "border-indigo-500 bg-indigo-600 text-white" : "border-slate-200 bg-white text-slate-600 hover:border-indigo-300 dark:border-slate-700 dark:bg-slate-800 dark:text-slate-300"}`}>{label}</button>
                  ))}
                </div>
                <p className="text-[11px] leading-5 text-slate-500 dark:text-slate-400">High keeps image detail best but may save less. Balanced is a middle ground. Smaller can soften photos and fine print more.</p>
              </div>
            </div>

            <div className="mt-4 flex gap-3 rounded-2xl border border-amber-100 bg-amber-50/80 p-4 dark:border-amber-900/50 dark:bg-amber-950/20">
              <Info className="mt-0.5 h-5 w-5 shrink-0 text-amber-700 dark:text-amber-300" />
              <div>
                <p className="text-sm font-bold text-amber-900 dark:text-amber-200">A sensible quality trade-off</p>
                <p className="mt-1 text-xs leading-5 text-amber-800 dark:text-amber-300">PDF text and vector content are kept where supported. Embedded images may be resized/recompressed; the more reduction you request, the more image detail can be lost. The tool reports the actual size achieved.</p>
              </div>
            </div>

            <button type="button" onClick={() => void compressPdf(false)} disabled={!file || isLoading || isCompressing} className="mt-5 inline-flex min-h-12 w-full items-center justify-center gap-2 rounded-xl bg-gradient-to-r from-[#5142c5] to-[#2878c8] px-5 py-3 text-sm font-bold text-white shadow-lg shadow-indigo-600/20 transition hover:-translate-y-0.5 hover:shadow-xl disabled:cursor-not-allowed disabled:opacity-45 disabled:hover:translate-y-0">
              {isCompressing ? <><Loader2 className="h-4 w-4 animate-spin" /> Compressing PDF…</> : <><Gauge className="h-4 w-4" /> Compress PDF</>}
            </button>
            {isCompressing && <p className="mt-3 text-center text-xs text-slate-500 dark:text-slate-400">{progressLabel || "Please keep this tab open while the PDF is compressed."}</p>}
          </section>
        </div>

        {error && <div role="alert" className="mt-6 flex items-start gap-3 rounded-2xl border border-rose-200 bg-rose-50 p-4 text-sm text-rose-800 dark:border-rose-900/60 dark:bg-rose-950/30 dark:text-rose-200"><Info className="mt-0.5 h-5 w-5 shrink-0" /><p>{error}</p></div>}

        {result && file && (
          <section className="mt-7 overflow-hidden rounded-3xl border border-emerald-200 bg-white shadow-sm dark:border-emerald-900/60 dark:bg-[#141a2c]">
            <div className="flex flex-col gap-4 bg-gradient-to-r from-emerald-50 to-teal-50 px-5 py-5 dark:from-emerald-950/30 dark:to-teal-950/20 sm:flex-row sm:items-center sm:justify-between sm:px-6">
              <div className="flex items-center gap-3">
                <div className="grid h-11 w-11 place-items-center rounded-xl bg-emerald-100 text-emerald-700 dark:bg-emerald-900/60 dark:text-emerald-300"><CheckCircle2 className="h-6 w-6" /></div>
                <div><h2 className="font-bold">Compression finished</h2><p className="mt-0.5 text-xs text-slate-600 dark:text-slate-300">{result.message}</p></div>
              </div>
              <button type="button" onClick={downloadResult} className="inline-flex shrink-0 items-center justify-center gap-2 rounded-xl bg-emerald-600 px-5 py-3 text-sm font-bold text-white transition hover:bg-emerald-700"><Download className="h-4 w-4" /> Download PDF</button>
            </div>
            <div className="grid gap-3 p-5 sm:grid-cols-4 sm:p-6">
              <div className="rounded-2xl bg-slate-50 p-4 dark:bg-slate-900/60"><p className="text-xs font-semibold text-slate-500">Original size</p><p className="mt-1 text-xl font-bold tabular-nums">{formatBytes(file.size)}</p></div>
              <div className="rounded-2xl bg-indigo-50 p-4 dark:bg-indigo-950/30"><p className="text-xs font-semibold text-indigo-700 dark:text-indigo-300">Compressed size</p><p className="mt-1 text-xl font-bold tabular-nums">{formatBytes(result.outputSize)}</p></div>
              <div className="rounded-2xl bg-emerald-50 p-4 dark:bg-emerald-950/30"><p className="text-xs font-semibold text-emerald-700 dark:text-emerald-300">Actual reduction</p><p className="mt-1 text-xl font-bold tabular-nums text-emerald-700 dark:text-emerald-300">{result.savedPercent.toFixed(1)}%</p><p className="mt-0.5 text-xs text-emerald-700/80 dark:text-emerald-400">{formatBytes(result.savedBytes)} saved</p></div>
              <div className="rounded-2xl bg-violet-50 p-4 dark:bg-violet-950/30"><p className="text-xs font-semibold text-violet-700 dark:text-violet-300">Quality profile</p><p className="mt-1 text-lg font-bold text-violet-800 dark:text-violet-200">{result.profileLabel}</p><p className="mt-0.5 text-xs text-violet-700/80 dark:text-violet-300">Image quality setting: {Math.round(result.imageQuality * 100)}%</p></div>
            </div>
            {!result.reachedTarget && (
              <div className="mx-5 mb-5 flex flex-col gap-3 rounded-xl border border-amber-200 bg-amber-50 px-4 py-3 text-xs leading-5 text-amber-800 dark:border-amber-900/50 dark:bg-amber-950/20 dark:text-amber-200 sm:mx-6 sm:flex-row sm:items-center sm:justify-between">
                <div><p className="font-bold">Target not fully reached</p><p>Requested: {targetPercent}% reduction · Actual: {result.savedPercent.toFixed(1)}%. You can keep this higher-quality result or try stronger image compression.</p></div>
                {!attemptedStronger && qualityMode !== "strong" && <button type="button" onClick={() => void compressPdf(true)} className="inline-flex shrink-0 items-center justify-center gap-2 rounded-lg bg-amber-700 px-4 py-2 font-bold text-white hover:bg-amber-800"><AlertTriangle className="h-4 w-4" /> Try stronger</button>}
              </div>
            )}
          </section>
        )}

        {file && pages.length > 0 && (
          <section className="mt-7 rounded-3xl border border-slate-200 bg-white p-5 shadow-sm dark:border-slate-800 dark:bg-[#141a2c] sm:p-6">
            <div className="flex flex-col gap-2 sm:flex-row sm:items-end sm:justify-between">
              <div><p className="text-xs font-bold uppercase tracking-[0.18em] text-indigo-600 dark:text-indigo-300">Step 03</p><h2 className="mt-1 text-xl font-bold">Page preview</h2><p className="mt-1 text-sm text-slate-500 dark:text-slate-400">{pageCount} page{pageCount === 1 ? "" : "s"} · Preview only; compression works on the uploaded PDF.</p></div>
              <span className="inline-flex w-fit items-center gap-1.5 rounded-full bg-indigo-50 px-3 py-1.5 text-xs font-semibold text-indigo-700 dark:bg-indigo-950/40 dark:text-indigo-300"><FileCheck2 className="h-3.5 w-3.5" /> Original preview</span>
            </div>
            <div className="mt-5 grid grid-cols-2 gap-3 sm:grid-cols-3 md:grid-cols-4 xl:grid-cols-6">
              {pages.map((page) => (
                <div key={page.pageNumber} className="min-w-0 rounded-2xl border border-slate-200 bg-slate-50 p-2 dark:border-slate-700 dark:bg-slate-900/50">
                  <div className="flex aspect-[3/4] items-center justify-center overflow-hidden rounded-lg bg-white p-1 shadow-inner dark:bg-slate-800">
                    {/* eslint-disable-next-line @next/next/no-img-element */}
                    <img src={page.dataUrl} alt={`PDF page ${page.pageNumber} preview`} className="h-full w-full object-contain" loading="lazy" />
                  </div>
                  <div className="mt-2 flex items-center justify-between gap-1"><span className="text-xs font-semibold text-slate-700 dark:text-slate-200">Page {page.pageNumber}</span><span className="text-[10px] text-slate-400">Preview</span></div>
                </div>
              ))}
            </div>
          </section>
        )}

        <div className="mt-6 flex items-start gap-3 rounded-2xl border border-slate-200 bg-white p-4 text-xs leading-5 text-slate-500 dark:border-slate-800 dark:bg-[#141a2c] dark:text-slate-400">
          <LockKeyhole className="mt-0.5 h-4 w-4 shrink-0 text-indigo-500" />
          <p>Files are processed in your browser and are not intentionally uploaded to Convia. Compression can change embedded image detail. Rewriting a digitally signed PDF can invalidate its signature, and some complex forms, annotations or attachments may not be preserved exactly. Keep your original file for important documents.</p>
        </div>
      </div>
    </main>
  );
}
